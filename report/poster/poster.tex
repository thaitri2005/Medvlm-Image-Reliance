% ================================================================== 1: problem
\column{0.23}

\block{The problem}{
A medical VLM can answer correctly from priors without looking at the image. The \emph{consistent-but-dangerous}
taxonomy flags such answers:

\vspace{3mm}
\begin{center}\small\renewcommand{\arraystretch}{1.25}
\begin{tabular}{@{}lcc@{}}
 & \textbf{changes} & \textbf{same}\\[-1mm]
 & \muted{without image} & \muted{without image}\\
\midrule
\textbf{stable} under paraphrase & \textsc{Ideal} & \dng\\
\textbf{flips} under paraphrase & \textsc{Fragile} & \textsc{Worst}\\
\end{tabular}
\end{center}
\vspace{2mm}
\innerblock{}{Its reliance test is \textbf{removing the image}. Can it tell us \textbf{which model} relies on the
image more?}
}

\block{Two models, one question}{
\includegraphics[width=\linewidth]{../figures/fig0_teaser.png}
\figcap{Without the image MedGemma answers (\dng) and Gemma-3 asks for the image (\textsc{Ideal}). A wrong image
gives the opposite verdict.}
}

\block{Contributions}{
\begin{itemize}
\item Four reliance probes compared on the same questions, 3 medical/general pairs.
\item What image removal measures instead: \textbf{refusal} and \textbf{priors}.
\item A pre-registered test \textbf{outside the training data}.
\end{itemize}
}

% ================================================================== 2: method
\column{0.23}

\block{Method}{
\textbf{Three pairs}, each a medical model and its general base:\\
\med{MedGemma-4B}, \med{MedGemma-1.5-4B} vs.\ \gen{Gemma-3-4B}\\
\med{MedVLThinker-3B} vs.\ \gen{Qwen2.5-VL-3B}

\vspace{4mm}
\textbf{Data}: SLAKE (CT, MRI, X-ray), VQA-RAD, PathVQA; 3,365 questions, 907 images.

\vspace{4mm}
\textbf{Probes} change only the image; we count how often the yes/no answer changes.

\vspace{2mm}
{\small\renewcommand{\arraystretch}{1.2}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.36\linewidth}>{\raggedright\arraybackslash}p{0.58\linewidth}@{}}
\toprule
\textbf{Image removed} & image tokens omitted (the taxonomy's test)\\
Wrong image & another image, same modality\\
Black image & black image in the image slot\\
\textbf{Opposite-answer} & same question, an image where the answer is the opposite\\
``Unknown'' allowed & image removed, may decline\\
\bottomrule
\end{tabular}}

\vspace{4mm}
\textbf{Scoring}: P(yes) from the yes/no token logits in one forward pass; the yes+no mass detects refusals.
bf16, greedy, identical prompts; about 120,000 calls on one 16\,GB GPU.

\vspace{4mm}
\innerblock{}{\textbf{Pre-registered}: endpoints frozen before the main run; each later probe and model pair
registered (dated git tag) before it ran. 95\% CIs by image.}
}

\block{Who answers without the image?}{
\begin{center}\includegraphics[width=0.44\linewidth]{../figures/fig4_abstention.pdf}\end{center}
\figcap{Share of questions declined without the image. Forced to say yes or no (top), Gemma-3 declines far more;
allowed to say ``unknown'' (bottom), both decline on $\ge$93\%.}
}

% ================================================================== 3: finding 1
\column{0.31}

\block{Finding 1: image removal agrees with no in-format probe}{
\includegraphics[width=\linewidth]{../figures/fig10_by_dataset.pdf}
\figcap{Medical \mn{} general difference in how often a yes/no answer changes when the image is altered.}

\vspace{3mm}
\begin{tikzfigure}
\begin{tabular}{@{}c@{\hspace{1.4cm}}c@{\hspace{1.4cm}}c@{}}
\bignum{5 of 9} & \bignum{0 of 5} & \bignum{+13.7}\\[1mm]
\begin{minipage}[t]{0.28\linewidth}\centering\small cells where image removal separates the models\end{minipage} &
\begin{minipage}[t]{0.28\linewidth}\centering\small of those confirmed by an in-format probe\end{minipage} &
\begin{minipage}[t]{0.28\linewidth}\centering\small points more \dng{} for MedGemma-4B, image removed\end{minipage}\\
\end{tabular}
\end{tikzfigure}
}

\block{Why: what image removal picks up}{
\begin{minipage}[t]{0.48\linewidth}\vspace{0pt}
\textbf{Refusal.} Without an image, \gen{Gemma-3} asks for it on \textbf{29\%} of questions, \med{MedGemma-4B} on
18\%, \med{MedGemma-1.5} on 0.7\%. Its forced answers there are unrelated to its image answers, so they count
as ``image-reliant''.
\end{minipage}\hfill
\begin{minipage}[t]{0.48\linewidth}\vspace{0pt}
\textbf{Priors.} A blind guess that matches the image leaves the answer unchanged, so image removal calls it
blind. MedGemma's extra \dng{} answers are \textbf{all correct} (+14.1 correct, \mn0.4 wrong).
\end{minipage}
}

% ================================================================== 4: finding 2, take-aways
\column{0.23}

\block{Finding 2: the in-format advantage stays in the training data}{
Follows the opposite-answer image, medical \mn{} general (points):

\vspace{2mm}
\begin{center}\renewcommand{\arraystretch}{1.25}
\begin{tabular}{@{}lrr@{}}
\toprule
& \textbf{SLAKE} & \textbf{PathVQA}\\[-1mm]
& \muted{\small trained on} & \muted{\small held out}\\
\midrule
\med{MedGemma-4B} & +48.0 & +0.8\\
\med{MedGemma-1.5} & +36.5 & \mn5.6\\
\med{MedVLThinker} & \mn1.6 & \mn0.9\\
\bottomrule
\end{tabular}
\end{center}
\vspace{1mm}
The PathVQA test was \textbf{registered before it ran}. The MedGemma predictions failed, so ``MedGemma uses the
image more'' was \textbf{withdrawn}.
}

\block{The \dng{} share, corrected}{
\begin{itemize}
\item 36--42\% of yes/no answers look image-independent \textbf{by chance}.
\item Its excess is mostly \textbf{correct} answers.
\item Reliance flags errors at chance (AUROC \textbf{0.51}).
\end{itemize}
}

\block{Take-aways}{
\begin{enumerate}
\item Don't rank models by image removal.
\item Probe in format, with an opposite-answer image.
\item Test outside each model's training data.
\end{enumerate}
}

\block{Limits}{
\small Three pairs; the explanations are post hoc; the held-out test is pathology only; paraphrases LLM-judged
(AI-assisted 50-item check: 88\%). Not a clinical claim.
